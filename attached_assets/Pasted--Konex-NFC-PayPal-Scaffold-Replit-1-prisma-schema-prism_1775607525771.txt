## 5) `src/lib/paypal.ts`

```ts
import { env } from './env';

const baseUrl = env.PAYPAL_ENV === 'live'
  ? 'https://api-m.paypal.com'
  : 'https://api-m.sandbox.paypal.com';

async function getAccessToken(): Promise<string> {
  const auth = Buffer.from(`${env.PAYPAL_CLIENT_ID}:${env.PAYPAL_CLIENT_SECRET}`).toString('base64');

  const res = await fetch(`${baseUrl}/v1/oauth2/token`, {
    method: 'POST',
    headers: {
      Authorization: `Basic ${auth}`,
      'Content-Type': 'application/x-www-form-urlencoded',
    },
    body: 'grant_type=client_credentials',
  });

  if (!res.ok) {
    const text = await res.text();
    throw new Error(`PayPal auth failed: ${text}`);
  }

  const data = await res.json();
  return data.access_token;
}

export async function createPaypalOrder(input: {
  value: string;
  currency: string;
  description: string;
  customId: string;
}) {
  const accessToken = await getAccessToken();

  const res = await fetch(`${baseUrl}/v2/checkout/orders`, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${accessToken}`,
      'Content-Type': 'application/json',
      Prefer: 'return=representation',
    },
    body: JSON.stringify({
      intent: 'CAPTURE',
      purchase_units: [
        {
          custom_id: input.customId,
          description: input.description,
          amount: {
            currency_code: input.currency,
            value: input.value,
          },
        },
      ],
      application_context: {
        return_url: `${env.APP_BASE_URL}/checkout/success`,
        cancel_url: `${env.APP_BASE_URL}/checkout/cancel`,
        brand_name: 'Konex NFC',
        shipping_preference: 'NO_SHIPPING',
        user_action: 'PAY_NOW',
      },
    }),
  });

  if (!res.ok) {
    const text = await res.text();
    throw new Error(`PayPal create order failed: ${text}`);
  }

  return res.json();
}

export async function capturePaypalOrder(orderId: string) {
  const accessToken = await getAccessToken();

  const res = await fetch(`${baseUrl}/v2/checkout/orders/${orderId}/capture`, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${accessToken}`,
      'Content-Type': 'application/json',
      Prefer: 'return=representation',
    },
  });

  if (!res.ok) {
    const text = await res.text();
    throw new Error(`PayPal capture failed: ${text}`);
  }

  return res.json();
}

export async function verifyPaypalWebhook(args: {
  transmissionId: string;
  transmissionTime: string;
  certUrl: string;
  authAlgo: string;
  transmissionSig: string;
  webhookEvent: unknown;
}) {
  const accessToken = await getAccessToken();

  const res = await fetch(`${baseUrl}/v1/notifications/verify-webhook-signature`, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${accessToken}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      auth_algo: args.authAlgo,
      cert_url: args.certUrl,
      transmission_id: args.transmissionId,
      transmission_sig: args.transmissionSig,
      transmission_time: args.transmissionTime,
      webhook_id: env.PAYPAL_WEBHOOK_ID,
      webhook_event: args.webhookEvent,
    }),
  });

  if (!res.ok) {
    const text = await res.text();
    throw new Error(`PayPal webhook verify failed: ${text}`);
  }

  const data = await res.json();
  return data.verification_status === 'SUCCESS';
}
```

## 8) `src/app/api/paypal/create-order/route.ts`

```ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { PRODUCT_CATALOG } from '@/lib/catalog';
import { createOrderSchema } from '@/lib/validators';
import { createPaypalOrder } from '@/lib/paypal';

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const { sku } = createOrderSchema.parse(body);
    const product = PRODUCT_CATALOG[sku];

    const localDraft = await prisma.order.create({
      data: {
        paypalOrderId: `draft_${crypto.randomUUID()}`,
        status: 'draft',
        amount: product.value,
        currency: product.currency,
        sku: product.sku,
        productType: product.productType,
      },
    });

    const paypalOrder = await createPaypalOrder({
      value: product.value,
      currency: product.currency,
      description: product.name,
      customId: localDraft.id,
    });

    await prisma.order.update({
      where: { id: localDraft.id },
      data: {
        paypalOrderId: paypalOrder.id,
        status: 'created',
      },
    });

    return NextResponse.json({ paypalOrderId: paypalOrder.id });
  } catch (error) {
    console.error(error);
    return NextResponse.json({ error: 'Unable to create order' }, { status: 400 });
  }
}
```

## 9) `src/app/api/paypal/capture-order/route.ts`

```ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { captureOrderSchema } from '@/lib/validators';
import { capturePaypalOrder } from '@/lib/paypal';
import { generateActivationToken } from '@/lib/token';

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const { orderID } = captureOrderSchema.parse(body);

    const order = await prisma.order.findUnique({
      where: { paypalOrderId: orderID },
      include: { activationToken: true, profile: true, card: true },
    });

    if (!order) {
      return NextResponse.json({ error: 'Order not found' }, { status: 404 });
    }

    if (['captured', 'activation_pending', 'activated'].includes(order.status)) {
      return NextResponse.json({
        ok: true,
        activationToken: order.activationToken?.token ?? null,
      });
    }

    const capture = await capturePaypalOrder(orderID);
    const captureInfo = capture?.purchase_units?.[0]?.payments?.captures?.[0];

    if (!captureInfo || captureInfo.status !== 'COMPLETED') {
      await prisma.order.update({
        where: { id: order.id },
        data: { status: 'failed' },
      });
      return NextResponse.json({ error: 'Capture not completed' }, { status: 400 });
    }

    const activationToken = generateActivationToken();

    const result = await prisma.$transaction(async (tx) => {
      const profile = await tx.profile.create({
        data: {
          orderId: order.id,
          publicSlug: `konex-${crypto.randomUUID().slice(0, 8)}`,
          theme: 'minimal',
          isLive: false,
        },
      });

      const token = await tx.activationToken.create({
        data: {
          orderId: order.id,
          profileId: profile.id,
          token: activationToken,
          expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
        },
      });

      if (order.productType === 'physical_card') {
        await tx.card.create({
          data: {
            orderId: order.id,
            sku: order.sku,
            status: 'activation_pending',
          },
        });
      }

      await tx.order.update({
        where: { id: order.id },
        data: {
          paypalCaptureId: captureInfo.id,
          status: 'activation_pending',
        },
      });

      return { token };
    });

    return NextResponse.json({ ok: true, activationToken: result.token.token });
  } catch (error) {
    console.error(error);
    return NextResponse.json({ error: 'Unable to capture order' }, { status: 400 });
  }
}
```

## 10) `src/app/api/activation/consume/route.ts`

```ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { activationSchema } from '@/lib/validators';

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const parsed = activationSchema.parse(body);

    const tokenRecord = await prisma.activationToken.findUnique({
      where: { token: parsed.token },
      include: {
        order: true,
        profile: true,
      },
    });

    if (!tokenRecord) {
      return NextResponse.json({ error: 'Invalid activation token' }, { status: 404 });
    }

    if (tokenRecord.consumedAt) {
      return NextResponse.json({ error: 'Token already used' }, { status: 409 });
    }

    if (tokenRecord.expiresAt < new Date()) {
      return NextResponse.json({ error: 'Token expired' }, { status: 410 });
    }

    const result = await prisma.$transaction(async (tx) => {
      const profile = await tx.profile.update({
        where: { id: tokenRecord.profileId },
        data: {
          displayName: parsed.displayName,
          title: parsed.title || null,
          company: parsed.company || null,
          phone: parsed.phone || null,
          email: parsed.email,
          website: parsed.website || null,
          bio: parsed.bio || null,
          avatarUrl: parsed.avatarUrl || null,
          theme: parsed.theme,
          isLive: true,
        },
      });

      await tx.activationToken.update({
        where: { id: tokenRecord.id },
        data: { consumedAt: new Date() },
      });

      await tx.order.update({
        where: { id: tokenRecord.orderId },
        data: { status: 'activated' },
      });

      await tx.card.updateMany({
        where: { orderId: tokenRecord.orderId },
        data: {
          status: 'active',
          activatedAt: new Date(),
        },
      });

      return profile;
    });

    return NextResponse.json({ ok: true, slug: result.publicSlug });
  } catch (error) {
    console.error(error);
    return NextResponse.json({ error: 'Activation failed' }, { status: 400 });
  }
}
```

## 11) `src/app/api/paypal/webhook/route.ts`

```ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { verifyPaypalWebhook } from '@/lib/paypal';

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();

    const isValid = await verifyPaypalWebhook({
      transmissionId: req.headers.get('paypal-transmission-id') || '',
      transmissionTime: req.headers.get('paypal-transmission-time') || '',
      certUrl: req.headers.get('paypal-cert-url') || '',
      authAlgo: req.headers.get('paypal-auth-algo') || '',
      transmissionSig: req.headers.get('paypal-transmission-sig') || '',
      webhookEvent: body,
    });

    if (!isValid) {
      return NextResponse.json({ error: 'Invalid webhook' }, { status: 400 });
    }

    const existing = await prisma.webhookEvent.findUnique({
      where: { paypalEventId: body.id },
    });

    if (existing) {
      return NextResponse.json({ ok: true, duplicate: true });
    }

    await prisma.webhookEvent.create({
      data: {
        paypalEventId: body.id,
        eventType: body.event_type,
        payloadJson: body,
      },
    });

    if (body.event_type === 'PAYMENT.CAPTURE.COMPLETED') {
      const captureId = body.resource?.id;
      if (captureId) {
        await prisma.order.updateMany({
          where: {
            paypalCaptureId: captureId,
            status: { in: ['captured', 'activation_pending'] },
          },
          data: { status: 'activation_pending' },
        });
      }
    }

    if (
      body.event_type === 'PAYMENT.CAPTURE.DENIED' ||
      body.event_type === 'CHECKOUT.PAYMENT-APPROVAL.REVERSED'
    ) {
      const reversedOrderId = body.resource?.supplementary_data?.related_ids?.order_id;
      if (reversedOrderId) {
        await prisma.order.updateMany({
          where: { paypalOrderId: reversedOrderId },
          data: { status: 'failed' },
        });
      }
    }

    return NextResponse.json({ ok: true });
  } catch (error) {
    console.error(error);
    return NextResponse.json({ error: 'Webhook handling failed' }, { status: 400 });
  }
}
```

## 13) `src/components/PayPalButton.tsx`

```tsx
'use client';

import { useEffect, useRef } from 'react';

declare global {
  interface Window {
    paypal: any;
  }
}

export function PayPalButton({ sku }: { sku: string }) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!window.paypal || !ref.current) return;

    ref.current.innerHTML = '';

    window.paypal.Buttons({
      async createOrder() {
        const res = await fetch('/api/paypal/create-order', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ sku }),
        });
        const data = await res.json();
        if (!res.ok) throw new Error(data.error || 'Create order failed');
        return data.paypalOrderId;
      },
      async onApprove(data: { orderID: string }) {
        const res = await fetch('/api/paypal/capture-order', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ orderID: data.orderID }),
        });
        const result = await res.json();
        if (!res.ok) throw new Error(result.error || 'Capture failed');
        window.location.href = `/activate?token=${result.activationToken}`;
      },
      onCancel() {
        window.location.href = '/checkout/cancel';
      },
      onError(err: unknown) {
        console.error(err);
        alert('Payment failed. Please try again.');
      },
    }).render(ref.current);
  }, [sku]);

  return <div ref={ref} />;
}
```

## 17) `src/app/p/[slug]/page.tsx`

```tsx
import { prisma } from '@/lib/prisma';
import { notFound } from 'next/navigation';

export default async function PublicProfilePage({ params }: { params: Promise<{ slug: string }> }) {
  const { slug } = await params;

  const profile = await prisma.profile.findUnique({
    where: { publicSlug: slug },
  });

  if (!profile || !profile.isLive) notFound();

  return (
    <main className="min-h-screen bg-black text-white px-6 py-10">
      <div className="mx-auto max-w-md rounded-3xl border border-white/10 bg-white/5 p-8">
        <h1 className="text-3xl font-semibold">{profile.displayName}</h1>
        {profile.title ? <p className="mt-2 text-white/70">{profile.title}</p> : null}
        {profile.company ? <p className="text-white/50">{profile.company}</p> : null}
        {profile.bio ? <p className="mt-6 text-white/80">{profile.bio}</p> : null}

        <div className="mt-6 grid gap-3">
          {profile.phone ? <a href={`tel:${profile.phone}`} className="rounded-xl bg-white text-black px-4 py-3">Call</a> : null}
          {profile.email ? <a href={`mailto:${profile.email}`} className="rounded-xl border border-white/20 px-4 py-3">Email</a> : null}
          {profile.website ? <a href={profile.website} className="rounded-xl border border-white/20 px-4 py-3">Website</a> : null}
        </div>
      </div>
    </main>
  );
}
```
