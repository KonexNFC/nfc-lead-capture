'use client';

import React, { useState } from 'react';

type KonexProfile = {
  firstName: string;
  lastName: string;
  company?: string;
  title?: string;
  phoneE164: string; // e.g. +447949483076
  email?: string;
  website?: string;
  avatarUrl?: string;
};

type Props = {
  profile: KonexProfile;
};

export default function KonexSaveContactCTA({ profile }: Props) {
  const [status, setStatus] = useState<'idle' | 'success' | 'error'>('idle');

  const generateVCard = () => {
    const lines = [
      'BEGIN:VCARD',
      'VERSION:3.0',
      `N:${profile.lastName};${profile.firstName};;;`,
      `FN:${profile.firstName} ${profile.lastName}`,
      profile.company ? `ORG:${profile.company}` : '',
      profile.title ? `TITLE:${profile.title}` : '',
      `TEL;TYPE=CELL:${profile.phoneE164}`,
      profile.email ? `EMAIL;TYPE=INTERNET:${profile.email}` : '',
      profile.website ? `URL:${profile.website}` : '',
      'END:VCARD',
    ].filter(Boolean);

    return lines.join('\n');
  };

  const downloadVCard = () => {
    const vcard = generateVCard();
    const blob = new Blob([vcard], { type: 'text/vcard;charset=utf-8' });
    const url = URL.createObjectURL(blob);

    const a = document.createElement('a');
    a.href = url;
    a.download = `${profile.firstName}-${profile.lastName}.vcf`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  };

  const handleSaveContact = async () => {
    try {
      // Web environment fallback using vCard download
      downloadVCard();
      setStatus('success');

      setTimeout(() => setStatus('idle'), 1800);
    } catch (e) {
      console.error('Save contact failed', e);
      setStatus('error');
    }
  };

  return (
    <div style={{ width: '100%', textAlign: 'center' }}>
      <button
        onClick={handleSaveContact}
        style={{
          width: '100%',
          padding: '14px 18px',
          borderRadius: 12,
          border: 'none',
          fontSize: 16,
          fontWeight: 600,
          cursor: 'pointer',
          background: 'linear-gradient(135deg, #1EA1FF, #3B82F6)',
          color: '#fff',
        }}
      >
        Add to Contacts
      </button>

      {status === 'success' && (
        <div
          style={{
            marginTop: 10,
            padding: '8px 12px',
            borderRadius: 8,
            background: 'rgba(34,197,94,0.15)',
            color: '#22c55e',
            fontSize: 14,
            fontWeight: 500,
          }}
        >
          Contact saved.
        </div>
      )}

      {status === 'error' && (
        <div
          style={{
            marginTop: 10,
            padding: '8px 12px',
            borderRadius: 8,
            background: 'rgba(239,68,68,0.15)',
            color: '#ef4444',
            fontSize: 14,
            fontWeight: 500,
          }}
        >
          Couldn’t save contact. Please try again.
        </div>
      )}
    </div>
  );
}
